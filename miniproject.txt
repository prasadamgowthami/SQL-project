CREATE TABLE Customers (
customer_id INT PRIMARY KEY,
name VARCHAR(100) NOT NULL,
email VARCHAR(100) UNIQUE,
phone VARCHAR(15),
city VARCHAR(255),
registration_date DATE
);

CREATE TABLE Categories (
category_id INT PRIMARY KEY,
category_name VARCHAR(100) NOT NULL,
description VARCHAR(255)
);

CREATE TABLE Products (
product_id INT PRIMARY KEY,
product_name VARCHAR(100) NOT NULL,
price DECIMAL(10,2),
stock_quantity INT,
category_id INT,
added_date DATE,
FOREIGN KEY (category_id) REFERENCES Categories(category_id)
);


CREATE TABLE Orders (
order_id INT PRIMARY KEY,
customer_id INT,
order_date DATE,
total_amount DECIMAL(10,2),
status VARCHAR(20),
FOREIGN KEY (customer_id) REFERENCES Customers(customer_id)
);


CREATE TABLE Order_Items (
order_item_id INT PRIMARY KEY,
order_id INT,
product_id INT,
quantity INT,
price DECIMAL(10,2),
subtotal DECIMAL(10,2),
FOREIGN KEY (order_id) REFERENCES Orders(order_id),
FOREIGN KEY (product_id) REFERENCES Products(product_id)
);

#Show customer name with their orders.
mysql> SELECT c.name, o.order_id, o.order_date, o.total_amount, o.status FROM Customers c JOIN Orders o ON c.customer_id = o.customer_id;

#Display product name with order quantity.
mysql> select p.*,oi.quantity from products p join Order_Items oi on p.product_id=oi.product_id;

#Show order details with customer information.
mysql> select o.order_id,o.order_date,o.total_amount,o.status,c.name,c.email,c.city FROM Orders o JOIN Customers c ON o.customer_id = c.customer_id;

#Show customers with products they purchased.
mysql> select c.name,p.product_name,oi.quantity from Customers c join Orders o on c.customer_id = o.customer_id join Order_Items oi ON o.order_id = oi.order_id JOIN Products p ON oi.product_id = p.product_id;

#Find customers who spent more than average spending.
mysql> SELECT c.customer_id,c.name,SUM(o.total_amount) AS total_spending FROM Customers c JOIN Orders o ON c.customer_id = o.customer_id WHERE o.status = 'Completed' GROUP BY c.customer_id, c.name HAVING SUM(o.total_amount) > (SELECT AVG(total_amount) FROM Orders WHERE status = 'Completed');

#Show orders above average total amount.

#Create a view to display customer orders
mysql> CREATE VIEW customer_orders AS SELECT c.customer_id,c.name,o.order_id,o.order_date,o.total_amount,o.status FROM Customers c JOIN Orders o ON c.customer_id = o.customer_id;
mysql> select * from customer_orders;

#Create a view for product details with category
mysql> create view product_details as (select p.*,c.category_name from products p join categories c on p.category_id=c.category_id);
mysql> select * from product_details;

#Create a view showing order item details
mysql> create view order_item_details as select i.order_item_id,i.order_id,p.product_name,i.quantity,i.price,i.subtotal from order_items i join products p on i.product_id=p.product_id;
mysql> select * from order_item_details;

#Create a view to display total spending by each customer
mysql> create view customer_total_spending as select c.customer_id,c.name,sum(o.total_amount) as total_spending from customers c join orders o on c.customer_id=o.customer_id group by c.customer_id,c.name;
mysql> select * from customer_total_spending;

#Create a view showing best-selling products
mysql> create view selling_products as select p.product_id,p.product_name,sum(i.quantity) as total_quantity_sold from products p join order_items i on p.product_id=i.product_id group by p.product_id,p.product_name;
mysql> select * from selling_products;

#Top 5 best selling products.
mysql> select p.product_id,p.product_name,sum(i.quantity) as total_quantity from products p join order_items i on p.product_id=i.product_id group by p.product_id,p.product_name order by total_quantity desc limit 5;

#Most frequent customers.
mysql> select c.customer_id,c.name,count(o.order_id) as total_orders from customers c join orders o on c.customer_id=o.customer_id group by c.customer_id,c.name order by total_orders desc;

#Monthly sales report.
 mysql> select year(order_date) as year,month(order_date) as month,sum(total_amount) as monthly_sales from orders group by year(order_date),month(order_date) order by year,month;

#Category generating highest revenue.
mysql> select c.category_id,c.category_name,sum(i.subtotal) as total_revenue from categories c join products p on c.category_id=p.category_id join order_items i on p.product_id =i.product_id group by c.category_id ,c.category_name order by total_revenue desc limit 1;

#Customer spending analysis.
mysql> select c.customer_id,c.name,count(o.order_id) as total_orders,coalesce(sum(o.total_amount),0) as total_spending,coalesce(avg(o.total_amount),0) as average_order_value from customers c left join orders o on c.customer_id=o.customer_id group by c.customer_id,c.name order by total_spending desc;
